  <script>

    /*

     * ============================================================

     * STUDENT DATA

     * ============================================================

     *

     * The order here must match the target order in

     * exhibition-targets.mind.

     */


    const students = [

      {

        name: "Student 1",

        title: "Object One",

        description: "Description of Student 1's object.",

        model: "./models/student-01.glb",


        // Adjust these three values when testing.

        scale: 0.20,

        position: [0, 0, 0],

        rotation: [0, 0, 0]

      },


      {

        name: "Student 2",

        title: "Object Two",

        description: "Description of Student 2's object.",

        model: "./models/student-02.glb",

        scale: 0.20,

        position: [0, 0, 0],

        rotation: [0, 0, 0]

      },


      {

        name: "Student 3",

        title: "Object Three",

        description: "Description of Student 3's object.",

        model: "./models/student-03.glb",

        scale: 0.20,

        position: [0, 0, 0],

        rotation: [0, 0, 0]

      }

    ];


    let mindarThree;

    let anchors = [];

    let mixers = [];

    let actions = [];


    let activeTarget = null;

    let animationPaused = false;


    const clock = new THREE.Clock();


    const startScreen =

      document.querySelector("#start-screen");


    const errorScreen =

      document.querySelector("#error-screen");


    const fallbackScreen =

      document.querySelector("#fallback-screen");


    const status =

      document.querySelector("#status");


    const objectTitle =

      document.querySelector("#object-title");


    const objectDescription =

      document.querySelector("#object-description");


    const errorMessage =

      document.querySelector("#error-message");


    const fallbackList =

      document.querySelector("#fallback-list");


    const startButton =

      document.querySelector("#start-button");


    const retryButton =

      document.querySelector("#retry-button");


    const pauseButton =

      document.querySelector("#pause-button");


    const restartButton =

      document.querySelector("#restart-button");


    /*

     * ============================================================

     * FALLBACK INFORMATION

     * ============================================================

     */


    function createFallbackInformation() {

      students.forEach((student, index) => {

        const article = document.createElement("article");


        article.innerHTML = `

          <strong>

            ${index + 1}. ${student.name}: ${student.title}

          </strong>

          <p>${student.description}</p>

        `;


        fallbackList.appendChild(article);

      });

    }


    createFallbackInformation();


    /*

     * ============================================================

     * USER INTERFACE

     * ============================================================

     */


    function showObjectInformation(index) {

      const student = students[index];


      activeTarget = index;


      objectTitle.textContent =

        `${student.name}: ${student.title}`;


      objectDescription.textContent =

        student.description;


      objectTitle.style.display = "block";

      objectDescription.style.display = "block";


      status.textContent =

        "Object detected — animation playing.";

    }


    function hideObjectInformation() {

      activeTarget = null;


      objectTitle.textContent = "";

      objectDescription.textContent = "";


      objectTitle.style.display = "none";

      objectDescription.style.display = "none";


      status.textContent =

        "Point your phone at a visual trigger.";

    }


    function showFallback() {

      fallbackScreen.classList.remove("hidden");

    }


    function hideFallback() {

      fallbackScreen.classList.add("hidden");

    }


    /*

     * ============================================================

     * CREATE AR SCENE

     * ============================================================

     */


    function createScene() {

      mindarThree = new MindARThree({

        container: document.querySelector("#ar-container"),


        /*

         * This file must contain three image targets.

         */

        imageTargetSrc: "./exhibition-targets.mind",


        /*

         * Allows the system to track up to three targets.

         */

        maxTrack: 3

      });


      const renderer = mindarThree.renderer;

      const scene = mindarThree.scene;

      const camera = mindarThree.camera;


      renderer.outputColorSpace = THREE.SRGBColorSpace;


      renderer.setPixelRatio(

        Math.min(window.devicePixelRatio || 1, 2)

      );


      /*

       * Lighting for the 3D models.

       */

      const hemisphereLight = new THREE.HemisphereLight(

        0xffffff,

        0x444444,

        2

      );


      scene.add(hemisphereLight);


      const directionalLight = new THREE.DirectionalLight(

        0xffffff,

        2

      );


      directionalLight.position.set(1, 3, 2);

      scene.add(directionalLight);


      /*

       * Create one anchor for each of the three targets.

       */

      anchors = students.map((student, index) => {

        const anchor = mindarThree.addAnchor(index);


        anchor.onTargetFound = () => {

          showObjectInformation(index);

        };


        anchor.onTargetLost = () => {

          if (activeTarget === index) {

            hideObjectInformation();

          }

        };


        return anchor;

      });


      loadModels();


      /*

       * Render loop.

       */

      renderer.setAnimationLoop(() => {

        const delta = clock.getDelta();


        if (!animationPaused) {

          mixers.forEach((mixer) => {

            if (mixer) {

              mixer.update(delta);

            }

          });

        }


        renderer.render(scene, camera);

      });

    }


    /*

     * ============================================================

     * LOAD THREE GLB MODELS

     * ============================================================

     */


    function loadModels() {

      const loader = new THREE.GLTFLoader();


      mixers = new Array(students.length).fill(null);

      actions = new Array(students.length).fill(null);


      students.forEach((student, index) => {

        loader.load(

          student.model,


          (gltf) => {

            const model = gltf.scene;


            model.scale.set(

              student.scale,

              student.scale,

              student.scale

            );


            model.position.set(

              student.position[0],

              student.position[1],

              student.position[2]

            );


            model.rotation.set(

              student.rotation[0],

              student.rotation[1],

              student.rotation[2]

            );


            anchors[index].group.add(model);


            /*

             * Play the first animation in the GLB file.

             */

            if (gltf.animations.length > 0) {

              const mixer = new THREE.AnimationMixer(model);

              const action = mixer.clipAction(gltf.animations[0]);


              action.play();


              mixers[index] = mixer;

              actions[index] = action;

            } else {

              console.warn(

                `${student.model} contains no animation.`

              );

            }

          },


          undefined,


          (error) => {

            console.error(

              `Could not load ${student.model}:`,

              error

            );


            status.textContent =

              `Could not load ${student.name}'s model.`;

          }

        );

      });

    }


    /*

     * ============================================================

     * START CAMERA

     * ============================================================

     */


    async function startAR() {

      startButton.disabled = true;

      retryButton.disabled = true;


      status.textContent = "Starting camera…";


      try {

        if (!mindarThree) {

          createScene();

        }


        await mindarThree.start();


        startScreen.classList.add("hidden");

        errorScreen.classList.add("hidden");


        status.textContent =

          "Camera active. Point at a visual trigger.";

      } catch (error) {

        console.error(error);


        let message =

          "Camera access was blocked or unavailable.";


        if (

          location.protocol !== "https:" &&

          location.hostname !== "localhost"

        ) {

          message +=

            "\n\nThe website must use HTTPS.";

        }


        if (error.message) {

          message += `\n\n${error.message}`;

        }


        errorMessage.textContent = message;

        errorScreen.classList.remove("hidden");

      } finally {

        startButton.disabled = false;

        retryButton.disabled = false;

      }

    }


    /*

     * ============================================================

     * ANIMATION BUTTONS

     * ============================================================

     */


    pauseButton.addEventListener("click", () => {

      animationPaused = !animationPaused;


      pauseButton.textContent = animationPaused

        ? "Play animation"

        : "Pause animation";


      status.textContent = animationPaused

        ? "Animation paused."

        : "Animation playing.";

    });


    restartButton.addEventListener("click", () => {

      actions.forEach((action) => {

        if (action) {

          action.reset();

          action.play();

        }

      });


      animationPaused = false;

      pauseButton.textContent = "Pause animation";

      status.textContent = "Animations restarted.";

    });


    /*

     * ============================================================

     * BUTTON EVENTS

     * ============================================================

     */


    startButton.addEventListener("click", startAR);

    retryButton.addEventListener("click", startAR);


    document

      .querySelector("#show-fallback-button")

      .addEventListener("click", showFallback);


    document

      .querySelector("#error-fallback-button")

      .addEventListener("click", showFallback);


    document

      .querySelector("#close-fallback-button")

      .addEventListener("click", hideFallback);

  </script>
